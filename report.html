24static std::string escapeHTML(const std::string &data) {
25 std::string buffer;
26 buffer.reserve(data.size() * 1.1);
27 for (size_t pos = 0; pos != data.size(); ++pos) {
28 switch (data[pos]) {
29 case '&':
30 buffer.append("&amp;");
31 break;
32 case '\"':
33 buffer.append("&quot;");
34 break;
35 case '\'':
36 buffer.append("&apos;");
37 break;
38 case '<':
39 buffer.append("&lt;");
40 break;
41 case '>':
42 buffer.append("&gt;");
43 break;
44 default:
45 buffer.append(&data[pos], 1);
46 break;
47 }
48 }
49 return buffer;
50}

52static std::unordered_set<std::string> extractLines(const std::string &text) {
53 std::unordered_set<std::string> lines;
54 std::istringstream ss(text);
55 std::string line;
56 while (std::getline(ss, line)) {
57 auto start = line.find_first_not_of(" \t\r\n");
58 if (start != std::string::npos) {
59 auto end = line.find_last_not_of(" \t\r\n");
60 std::string t = line.substr(start, end - start + 1);
61 if (t.length() > 3)
62 lines.insert(t);
63 }
64 }
65 return lines;
66}

421// Helper: Get all supported files from directory
422std::vector<std::string> getFilesFromDirectory(const std::string &dirPath) {
423 std::vector<std::string> files;
424 if (!fs::is_directory(dirPath)) {
425 std::cerr << "Error: " << dirPath << " is not a valid directory." << std::endl;
426 return files;
427 }
428
429 for (const auto &entry : fs::directory_iterator(dirPath)) {
430 if (entry.is_regular_file()) {
431 std::string ext = entry.path().extension().string();
432 if (ext == ".cpp" || ext == ".h" || ext == ".txt" || ext == ".c" ||
433 ext == ".cc" || ext == ".cxx") {
434 files.push_back(entry.path().string());
435 }
436 }
437 }
438
439 if (files.empty()) {
440 std::cerr << "No supported files (.cpp, .h, .txt, .c) found in " << dirPath
441 << std::endl;
442 }
443
444 return files;
445}
